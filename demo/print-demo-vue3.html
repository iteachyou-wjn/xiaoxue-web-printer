一、引入 SDK

将 xiaoxue-printer-sdk.min.js 复制到项目，并在 index.html 中引入：

<script src="path/to/xiaoxue-printer-sdk.min.js"></script>

二、在 Vue3 组件中使用（组合式 API）

// App.vue  setup() 中
<script setup>
import { ref, onMounted, onUnmounted } from 'vue'

const sdk = new DreamerPrinterSDK({ url: 'ws://localhost:54321' })
const connected = ref(false)
const printers = ref([])
const paperSizes = ref([])

onMounted(() => {
  sdk.on('open', () => { connected.value = true; fetchPrinters() })
  sdk.connect()
})
onUnmounted(() => sdk.disconnect())

async function fetchPrinters() {
  const r = await sdk.getPrinters()
  printers.value = r.printers || []
}
async function fetchPageSize(p) {
  const r = await sdk.getPageSize(p)
  paperSizes.value = r.sizes || []
}
// 打印
function doPrint() {
  sdk.doPrint({
    printer: '默认打印机',
    style: { paper: 'A4', direction: 'vertical', fontSize: 12 },
    content: { type: 'text', value: '打印内容' }
  }).then(r => console.log('任务ID:', r.taskId))
}
</script>

三、API 一览

  1. getPrinters() - 获取可用打印机
  2. getPageSize(printerName) - 获取指定打印机纸张
  3. doPrint(json) - 提交并执行打印任务
  4. doPreview(json) - 打开打印预览窗口
  5. getTaskList / getTaskStatus / cancelTask / clearTasks - 任务管理
  6. getBartenderInstance() - 获取 BarTender 实例（含版本号）

// 获取 BarTender 实例（含版本号）
const bt = await sdk.getBartenderInstance()

一、任务管理 API

以下接口均需通过 DreamerPrinterSDK 实例调用：

  1. getTaskList() - 获取任务列表，返回 { tasks: [...] }
  2. getTaskStatus(taskId) - 查询单个任务状态，返回 { task?, found }
  3. cancelTask(taskId) - 取消「待打印 / 打印中」任务，返回 { success, message }
  4. clearTasks() - 清除「已完成 / 失败」任务，返回 { success, message }
  5. doPrint(json) - 提交并执行打印任务，返回 { success, taskId }
  6. taskListUpdate 事件 - 服务端在任务变化时主动推送任务列表

二、在 Vue3 中使用（组合式 API）

<script setup>
import { ref, onMounted, onUnmounted } from 'vue'

const sdk = new DreamerPrinterSDK({ url: 'ws://localhost:54321' })
const tasks = ref([])

onMounted(() => {
  // 订阅任务列表变化（服务端主动推送）
  sdk.on('taskListUpdate', (list) => { tasks.value = list || [] })
  sdk.on('open', refresh)
  sdk.connect()
})
onUnmounted(() => sdk.disconnect())

async function refresh() {
  const r = await sdk.getTaskList()
  tasks.value = r.tasks || []
}
// 提交打印任务
function submit() {
  sdk.doPrint({ printer: '默认打印机', style: { paper: 'A4' },
    content: { type: 'text', value: '打印内容' } })
    .then(r => console.log('任务ID:', r.taskId))
}
// 查询状态 / 取消 / 清除
function query(id) { sdk.getTaskStatus(id).then(r => console.log(r.task)) }
function cancel(id) { sdk.cancelTask(id).then(r => console.log(r.message)) }
function clearDone() { sdk.clearTasks().then(r => console.log(r.message)) }
</script>

三、任务状态说明

  1. PENDING - 待打印（可取消）
  2. PRINTING - 打印中（可取消）
  3. COMPLETED - 已完成
  4. FAILED - 失败（可通过 errorMessage 查看原因）
