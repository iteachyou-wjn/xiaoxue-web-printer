文本打印示例（app.component.ts）

通过 PrintService（封装 SDK）提交 content.type='text' 的打印任务。完整接入步骤见「集成代码」tab。

import { Component, OnInit } from '@angular/core';
import { PrintService } from './print.service';

@Component({
  selector: 'app-root',
  templateUrl: './app.component.html'
})
export class AppComponent implements OnInit {
  printers: string[] = [];
  printer = '';
  content = '尊敬的客户：\n\n感谢您使用梦想家WEB打印控件（Angular 集成示例）。';

  constructor(private print: PrintService) {}

  ngOnInit() {
    this.print.connect();
    setTimeout(() => this.refreshPrinters(), 600);
  }

  refreshPrinters() {
    this.print.getPrinters().then((r: any) => { this.printers = r.printers || []; });
  }

  private buildTextRequest() {
    return {
      printer: this.printer,
      style: { paper: 'A4', direction: 'vertical', fontSize: 12 },
      content: { type: 'text', value: this.content }   // 文本打印
    };
  }

  doPreview() { this.print.preview(this.buildTextRequest()).then(() => alert('预览已发送')); }
  doPrint()   { this.print.print(this.buildTextRequest()).then((r) => alert('任务ID: ' + r.taskId)); }
}

第一步：引入 SDK

将 dreamer-printer-sdk.min.js 放入工程的 src/assets/ 目录，并在 index.html 中引入：

<!-- src/index.html -->
<script src="assets/dreamer-printer-sdk.min.js"></script>

若需在组件中使用 TS 类型，添加全局类型声明 src/typings.d.ts：

declare class DreamerPrinterSDK {
  constructor(options?: { url?: string; autoReconnect?: boolean });
  connect(): this;  disconnect(): void;
  on(event: string, cb: Function): this;
  getPrinters(): Promise<any>;
  getPageSize(printerName: string): Promise<any>;
  doPrint(json: any): Promise<any>;
  doPreview(json: any): Promise<any>;
  getTaskList(): Promise<any>;
  cancelTask(taskId: string): Promise<any>;
  clearTasks(): Promise<any>;
}

第二步：封装打印服务（print.service.ts）

import { Injectable } from '@angular/core';
declare const DreamerPrinterSDK: any;

@Injectable({ providedIn: 'root' })
export class PrintService {
  private sdk: any;
  constructor() {
    this.sdk = new DreamerPrinterSDK({ url: 'ws://localhost:54321' });
  }
  connect()      { this.sdk.connect(); }
  disconnect()   { this.sdk.disconnect(); }
  getPrinters()  { return this.sdk.getPrinters(); }
  getPageSize(p) { return this.sdk.getPageSize(p); }
  preview(json)  { return this.sdk.doPreview(json); }
  print(json)    { return this.sdk.doPrint(json); }
}

第三步：组件中使用与模板

文本 / HTML 两种场景的组件代码见上方两个 tab。模板示意：

<!-- app.component.html -->
<select [(ngModel)]="printer">
  <option *ngFor="let p of printers" [value]="p">{{ p }}</option>
</select>
<textarea [(ngModel)]="content"></textarea>
<button (click)="doPreview()">打印预览</button>
<button (click)="doPrint()">执行打印</button>

如需使用 ngModel，请在模块中引入 FormsModule。

要点

  1. SDK 通过全局脚本引入，组件用 declare const DreamerPrinterSDK 声明类型。
  2. 建议封装为 PrintService 单例（providedIn: 'root'），多组件复用同一连接。
  3. 所有 API 返回 Promise，需通过实例调用。
  4. 组件销毁时在 ngOnDestroy 中 disconnect()，避免连接泄漏。
  5. BarTender 模板参数 / 图像集成见「BarTender」标签页。

任务管理 API（Angular）

以下接口均需通过 DreamerPrinterSDK 实例调用：

  1. getTaskList() - 获取任务列表，返回 { tasks: [...] }
  2. getTaskStatus(taskId) - 查询单个任务状态，返回 { task?, found }
  3. cancelTask(taskId) - 取消「待打印 / 打印中」任务，返回 { success, message }
  4. clearTasks() - 清除「已完成 / 失败」任务，返回 { success, message }
  5. doPrint(json) - 提交并执行打印任务，返回 { success, taskId }
  6. taskListUpdate 事件 - 服务端在任务变化时主动推送任务列表

在 Angular 中使用（PrintService 扩展）

在 PrintService 中补充任务相关方法，并通过 BehaviorSubject 对外广播任务列表：

import { Injectable } from '@angular/core';
import { BehaviorSubject } from 'rxjs';
declare const DreamerPrinterSDK: any;

@Injectable({ providedIn: 'root' })
export class PrintService {
  private sdk: any;
  readonly tasks$ = new BehaviorSubject<any[]>([]);

  constructor() {
    this.sdk = new DreamerPrinterSDK({ url: 'ws://localhost:54321' });
    this.sdk.on('taskListUpdate', (list: any[]) => this.tasks$.next(list || []));
  }
  connect()      { this.sdk.connect(); }
  disconnect()   { this.sdk.disconnect(); }
  getTaskList()  { return this.sdk.getTaskList(); }
  getTaskStatus(id: string) { return this.sdk.getTaskStatus(id); }
  cancelTask(id: string) { return this.sdk.cancelTask(id); }
  clearTasks()   { return this.sdk.clearTasks(); }
  submitPrint(job: any) { return this.sdk.doPrint(job); }
}

任务状态说明

  1. PENDING - 待打印（可取消）
  2. PRINTING - 打印中（可取消）
  3. COMPLETED - 已完成
  4. FAILED - 失败（可通过 errorMessage 查看原因）

BarTender 集成（Angular）

通过 getBartenderInstance() 获取 BarTender 实例，再调用模板参数 / 图像接口。各方法返回 Promise：

  1. getBartenderInstance() - 获取实例，返回 { available, version, fullVersion }
  2. getTemplateParams(templatePath) - 获取模板参数，返回 { name, type }[]（type：Barcode/Text/Picture 等，条码不细分）
  3. getTemplateImage(templatePath) - 获取模板图像，返回 base64
  4. getTemplateImageWithParams(templatePath, params) - 获取带参数模板预览图像（base64）

在 Angular 中使用（PrintService 扩展）

// PrintService 中新增 BarTender 方法
getBartender() {
  return this.sdk.getBartenderInstance();
}
// 模板参数（含控件类型）
getTemplateParams(path: string) {
  return this.getBartender().then(bt => bt.getTemplateParams(path));
}
// 模板图像（base64）
getTemplateImage(path: string) {
  return this.getBartender().then(bt => bt.getTemplateImage(path));
}
// 带参数模板预览图像（base64）
getTemplateImageWithParams(path: string, params: any) {
  return this.getBartender().then(bt => bt.getTemplateImageWithParams(path, params));
}

BarTender 打印 / 预览

提交打印或预览时，将 content.type 设为 'bartender'，服务端将调用本机 BarTender 渲染/打印模板：

// app.component.ts
private buildBartenderRequest() {
  const params = JSON.parse(this.btParamsText || '{}');
  return {
    printer: this.printer,
    style: { paper: 'A4', direction: 'vertical' },
    content: {
      type: 'bartender',          // 关键：BarTender 类型
      value: this.btTemplate,     // 模板路径 .btw
      params: params              // 模板字段值
    }
  };
}

doPreview() { this.print.preview(this.buildBartenderRequest()); }  // 打印预览
doPrint()   { this.print.print(this.buildBartenderRequest()); }    // 执行打印
